# Storage menu (in the run's own storage directory, the same in every run): save setup (slot 1), save waveform CSV, save screenshot PNG, default set-up, recall the setup
key ch2
close
knob hscale -2
knob vscale +1
menu Storage
frames 2
shot $S/b0-storage-menu.ppm
do Save setup
frames 1
shot $S/b1-storage-setup-saved.ppm
do Save waveform (CSV)
frames 1
shot $S/b2-storage-csv-saved.ppm
do Save screenshot (PNG)
frames 2
shot $S/b3-storage-png-saved.ppm
key default
frames 5
shot $S/b4-storage-default.ppm
menu Storage
do Recall setup
frames 5
shot $S/b5-storage-recalled.ppm
